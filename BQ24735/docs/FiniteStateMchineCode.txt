//FiniteStateMachine_v2
digraph FSM {
    rankdir=LR;
    node [shape = circle, fontsize=12, fontname="Arial"];

    // States
    POWER_ON [shape=doublecircle, style=filled, fillcolor=lightyellow, label="POWER_ON\n"];
    INIT [label="SYSTEM\nINIT"];
    CHECK_MAINS [label="CHECK_MAINS"];
    SEND_I2C_CMD [label="SEND_I2C_CMD\n(One-time)"];
    CHARGE_BATTERY [label="CHARGE_BATTERY \n&\n GUNS POWER BY MAINS", style=filled, fillcolor=lightpink];
    POWER_GUNS_MAINS [label="GUNS\n", style=filled, fillcolor="lightgrey"];
    CHECK_AC_LOST [label="CHECK_AC_LOST"];
    BATTERY_POWER [label="BATTERY_POWER", style=filled, fillcolor= palegreen];

    // Transitions
    POWER_ON -> INIT [label=""];
    INIT -> CHECK_MAINS [label=""];

    // Mains present first time
    CHECK_MAINS -> SEND_I2C_CMD [label="PRESENT"];
    CHARGE_BATTERY -> POWER_GUNS_MAINS [label="Lock and unlock control"];
    POWER_GUNS_MAINS -> CHECK_MAINS [label=""];

    // Mains already present
    CHECK_MAINS -> CHARGE_BATTERY [label="PRESENT\n"];

    // Mains absent
    CHECK_MAINS -> CHECK_AC_LOST [label="\nNOT PRESENT"];
    CHECK_AC_LOST -> BATTERY_POWER [label="\n"];
    BATTERY_POWER -> POWER_GUNS_MAINS [label="\nUnlock Guns"];

}
